function ConfidenceContributors({ contributors }: { contributors: Record<string, number> }) {

function getNestedValue(obj: Record<string, unknown>, dotPath: string): number {
  const keys = dotPath.split('.');
  let cur: unknown = obj;
  for (const k of keys) {
    if (cur && typeof cur === 'object') cur = (cur as Record<string, unknown>)[k];
    else return 0;
  }
  return typeof cur === 'number' ? cur : 0;
}
